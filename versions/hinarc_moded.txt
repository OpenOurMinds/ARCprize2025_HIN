class BaseDataLoader:\
    """Base class for creating a tf.data.Dataset pipeline."""\
    def __init__(self, config):\
        self.config = config\
\
    def _pad_and_crop_grid_tf(self, grid_tensor):\
        height, width = tf.shape(grid_tensor)[0], tf.shape(grid_tensor)[1]\
        pad_height = tf.maximum(0, self.config.grid_size - height) #[cite: 137]\
        pad_width = tf.maximum(0, self.config.grid_size - width) #[cite: 137]\
        padded_grid = tf.pad(grid_tensor, [[0, pad_height], [0, pad_width]], constant_values=0) #[cite: 137]\
        cropped_grid = padded_grid[:self.config.grid_size, :self.config.grid_size] #[cite: 137]\
        cropped_grid.set_shape([self.config.grid_size, self.config.grid_size]) #[cite: 137]\
        return tf.cast(cropped_grid, dtype=tf.int32) #[cite: 137]\
\
    def _create_raw_dataset(self):\
        ds = tf.data.Dataset.from_generator(\
            self._get_data_generator,\
            output_signature=tf.TensorSpec(shape=(None, None), dtype=tf.int32)\
        ) #[cite: 138]\
        ds = ds.filter(lambda x: tf.rank(x) == 2 and tf.shape(x)[0] > 0 and tf.shape(x)[1] > 0) #[cite: 138]\
        ds = ds.map(self._pad_and_crop_grid_tf, num_parallel_calls=tf.data.AUTOTUNE) #[cite: 138]\
        ds = ds.map(lambda x: tf.expand_dims(tf.cast(x, tf.float32), axis=-1), num_parallel_calls=tf.data.AUTOTUNE) #[cite: 139]\
        return ds #[cite: 139]\
\
    def load_dataset(self):\
        ds = self._create_raw_dataset() #[cite: 140]\
        ds = ds.shuffle(buffer_size=5000) #[cite: 140]\
        ds = ds.batch(self.config.batch_size, drop_remainder=True) #[cite: 140]\
        ds = ds.prefetch(tf.data.AUTOTUNE) #[cite: 140]\
        return ds #[cite: 140]\

class SyntheticDataLoader(BaseDataLoader):\
    """Data loader for GAN-generated synthetic grids."""\
    def _get_data_generator(self):\
        for path in self.config.synthetic_data_paths:\
            if not os.path.exists(path):\
                logging.warning(f"File not found, skipping: \{path\}") #[cite: 145]\
                continue\
            with open(path, 'r') as f:\
                data = json.load(f) #[cite: 145]\
            for item in data:\
                grid = None\
                if isinstance(item, list) and all(isinstance(sublist, list) for sublist in item):\
                    grid = np.array(item, dtype=np.int32) #[cite: 146]\
                elif isinstance(item, list) and all(isinstance(val, (int, float)) for val in item):\
                    grid_size_sq = len(item) #[cite: 147]\
                    grid_size = int(np.sqrt(grid_size_sq)) #[cite: 147]\
                    if grid_size * grid_size == grid_size_sq:\
                        grid = np.array(item, dtype=np.int32).reshape(grid_size, grid_size) #[cite: 148]\
                if grid is not None and grid.size > 0:\
                    yield grid #[cite: 149]\

class Encoder(Model):\
    """Maps the input grid to a dense latent representation#[cite: 181]."""\
    def __init__(self, latent_dim, grid_size, name='encoder', **kwargs):\
        super(Encoder, self).__init__(name=name, **kwargs)\
        self.conv1 = layers.Conv2D(32, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal()) #[cite: 183]\
        self.bn1 = layers.BatchNormalization() #[cite: 183]\
        self.conv2 = layers.Conv2D(64, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal()) #[cite: 183]\
        self.bn2 = layers.BatchNormalization() #[cite: 183]\
        self.flatten = layers.Flatten() #[cite: 183]\
        self.dense = layers.Dense(latent_dim, name='dense_latent') #[cite: 184]\
\
    def call(self, inputs, training=None):\
        x = self.conv1(inputs) #[cite: 184]\
        x = self.bn1(x, training=training) #[cite: 184]\
        x = self.conv2(x) #[cite: 184]\
        x = self.bn2(x, training=training) #[cite: 184]\
        x = self.flatten(x) #[cite: 184]\
        x = self.dense(x) #[cite: 184]\
        return x #[cite: 184]\

class Decoder(Model):\
    """Reconstructs the grid from the quantized latent representation#[cite: 185]."""\
    def __init__(self, latent_dim, num_colors, grid_size, name='decoder', **kwargs):\
        super(Decoder, self).__init__(name=name, **kwargs)\
        self.grid_size = grid_size\
        self.num_colors = num_colors\
        initial_h, initial_w = 8, 8 #[cite: 187]\
        self.dense = layers.Dense(initial_h * initial_w * 64, activation='relu', name='dense_upsample') #[cite: 187]\
        self.reshape = layers.Reshape((initial_h, initial_w, 64)) #[cite: 187]\
        self.deconv1 = layers.Conv2DTranspose(64, 3, activation='relu', strides=2, padding='same') #[cite: 187]\
        self.bn1 = layers.BatchNormalization() #[cite: 187]\
        self.deconv2 = layers.Conv2DTranspose(32, 3, activation='relu', strides=2, padding='same') #[cite: 187]\
        self.bn2 = layers.BatchNormalization() #[cite: 188]\
        self.output_conv = layers.Conv2D(num_colors, 3, activation='softmax', padding='same') #[cite: 188]\
        self.cropping = layers.Cropping2D(cropping=((1, 1), (1, 1))) #[cite: 190]\
\
    def call(self, inputs, training=None):\
        x = self.dense(inputs) #[cite: 190]\
        x = self.reshape(x) #[cite: 190]\
        x = self.deconv1(x) #[cite: 190]\
        x = self.bn1(x, training=training) #[cite: 190]\
        x = self.deconv2(x) #[cite: 190]\
        x = self.bn2(x, training=training) #[cite: 190]\
        output_logits = self.output_conv(x) #[cite: 191]\
        final_output = self.cropping(output_logits) #[cite: 192]\
        final_output.set_shape([None, self.grid_size, self.grid_size, self.num_colors]) #[cite: 192]\
        return final_output #[cite: 192]\

class VectorQuantizer(layers.Layer):\
    """A Keras Layer for Vector Quantization."""\
    def __init__(self, num_embeddings, embedding_dim, beta=0.25, **kwargs):\
        super().__init__(**kwargs)\
        self.embedding_dim = embedding_dim\
        self.num_embeddings = num_embeddings\
        self.beta = beta #[cite: 193]\
        self.embeddings = self.add_weight(name="embeddings_vqvae", shape=[self.num_embeddings, self.embedding_dim], initializer=RandomUniform(), trainable=True) #[cite: 193]\
\
    def call(self, x):\
        flattened = tf.reshape(x, [-1, self.embedding_dim]) #[cite: 194]\
        distances = tf.reduce_sum(flattened**2, 1, keepdims=True) + tf.transpose(tf.reduce_sum(self.embeddings**2, 1, keepdims=True)) - 2 * tf.matmul(flattened, self.embeddings, transpose_b=True) #[cite: 194]\
        encoding_indices = tf.argmin(distances, axis=1) #[cite: 194]\
        quantized = tf.matmul(tf.one_hot(encoding_indices, self.num_embeddings), self.embeddings) #[cite: 195]\
        quantized = tf.reshape(quantized, tf.shape(x)) #[cite: 195]\
        commitment_loss = tf.reduce_mean((tf.stop_gradient(quantized) - x) ** 2) #[cite: 195]\
        dictionary_loss = tf.reduce_mean((quantized - tf.stop_gradient(x)) ** 2) #[cite: 195]\
        self.add_loss(self.beta * commitment_loss + dictionary_loss) #[cite: 195]\
        quantized = x + tf.stop_gradient(quantized - x) #[cite: 195]\
        return quantized #[cite: 196]\

class VQVAE(keras.Model):\
    """Chains the Encoder, VectorQuantizer, and Decoder together#[cite: 197]."""\
    def __init__(self, num_embeddings, embedding_dim, num_colors, grid_size, **kwargs):\
        super(VQVAE, self).__init__(**kwargs)\
        self.encoder = Encoder(latent_dim=embedding_dim, grid_size=grid_size) #[cite: 197]\
        self.vector_quantizer = VectorQuantizer(num_embeddings, embedding_dim) #[cite: 197]\
        self.decoder = Decoder(latent_dim=embedding_dim, num_colors=num_colors, grid_size=grid_size) #[cite: 197]\
        self.total_loss_tracker = tf.keras.metrics.Mean(name="total_loss") #[cite: 197]\
        self.reconstruction_loss_tracker = tf.keras.metrics.Mean(name="reconstruction_loss") #[cite: 197]\
        self.vq_loss_tracker = tf.keras.metrics.Mean(name="vq_loss") #[cite: 197]\
\
    def call(self, inputs):\
        encoded_outputs = self.encoder(inputs) #[cite: 198]\
        quantized_latents = self.vector_quantizer(encoded_outputs) #[cite: 198]\
        reconstructions = self.decoder(quantized_latents) #[cite: 198]\
        return reconstructions #[cite: 198]\
\
    def train_step(self, data):\
        with tf.GradientTape() as tape:\
            reconstructions = self(data, training=True) #[cite: 199]\
            reconstruction_loss = tf.reduce_mean(SparseCategoricalCrossentropy(from_logits=False)(tf.cast(data, tf.int32), reconstructions)) #[cite: 199]\
            vq_loss = sum(self.vector_quantizer.losses) #[cite: 200]\
            total_loss = reconstruction_loss + vq_loss #[cite: 200]\
        grads = tape.gradient(total_loss, self.trainable_variables) #[cite: 200]\
        self.optimizer.apply_gradients(zip(grads, self.trainable_variables)) #[cite: 200]\
        self.total_loss_tracker.update_state(total_loss) #[cite: 200]\
        self.reconstruction_loss_tracker.update_state(reconstruction_loss) #[cite: 200]\
        self.vq_loss_tracker.update_state(vq_loss) #[cite: 200]\
        return \{"loss": self.total_loss_tracker.result(), "reconstruction_loss": self.reconstruction_loss_tracker.result(), "vq_loss": self.vq_loss_tracker.result()\} #[cite: 201]\

# --- RL Agent and Solver ---\
class RLAgent:\
    """RL agent that generates plausible grids by optimizing a latent vector#[cite: 211]."""\
    def __init__(self, vae_model, config):\
        self.vae_model = vae_model\
        self.config = config\
        self.latent_vector = tf.Variable(tf.random.uniform(shape=(1, self.config.rl_latent_dim)), dtype=tf.float32, trainable=True) #[cite: 211, 212]\
        self.optimizer = Adam(learning_rate=self.config.rl_learning_rate) #[cite: 212]\
\
    @tf.function\
    def train_step(self):\
        with tf.GradientTape() as tape:\
            codebook = self.vae_model.vector_quantizer.embeddings #[cite: 214]\
            distances = tf.reduce_sum((self.latent_vector - codebook) ** 2, axis=1) #[cite: 215]\
            loss = tf.reduce_min(distances) #[cite: 216]\
        grads = tape.gradient(loss, [self.latent_vector]) #[cite: 216]\
        self.optimizer.apply_gradients(zip(grads, [self.latent_vector])) #[cite: 216]\
        return loss #[cite: 217]\
\
    def generate_and_train(self, max_steps):\
        logging.info("RL Agent: Starting latent vector optimization...") #[cite: 217]\
        for step in range(max_steps):\
            loss = self.train_step() #[cite: 217]\
            if tf.math.is_nan(loss):\
                logging.error(f"RL Agent: Training failed at step \{step\}. Loss is NaN.") #[cite: 218]\
                break\
            if (step + 1) % 1000 == 0:\
                logging.info(f"RL Agent: Step \{step + 1\}/\{max_steps\}, Loss: \{loss.numpy():.4f\}") #[cite: 218]\
        logging.info("RL Agent: Latent vector optimization complete.") #[cite: 219]\
        final_grid_logits = self.vae_model.decoder(self.latent_vector) #[cite: 219]\
        final_grid = tf.argmax(final_grid_logits, axis=-1, output_type=tf.int32).numpy().squeeze() #[cite: 219]\
        return final_grid #[cite: 219]\

def solve_task(task, solution_grid):\
    """\
    Placeholder solver that returns the pre-generated solution grid for all test cases.\
    """\
    # The model generates one solution; we apply it to all test inputs for a given task.\
    predictions = []\
    for _ in task['test']:\
        # Ensure the grid is a list of lists for JSON serialization\
        predictions.append(solution_grid.tolist())\
    return predictions\
