import torch
import torch.nn as nn
import torch.nn.functional as F

class TransformerDiscriminator(nn.Module):
    def __init__(self, vocab_size, hidden_size, num_layers, num_heads, max_seq_length, condition_size=0, num_channels=2):
        super(TransformerDiscriminator, self).__init__()
        self.vocab_size = vocab_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.num_heads = num_heads
        self.max_seq_length = max_seq_length
        self.condition_size = condition_size
        self.num_channels = num_channels # Number of channels (e.g., input and output)

        # Embedding layer for tokens
        self.embedding = nn.Embedding(vocab_size, hidden_size)

        # Positional Encoding (using learned embeddings)
        self.positional_embedding = nn.Embedding(max_seq_length, hidden_size)

        # Channel Embedding
        self.channel_embedding = nn.Embedding(num_channels, hidden_size)


        # If using conditioning, a linear layer to process it
        if self.condition_size > 0:
             self.condition_linear = nn.Linear(condition_size, hidden_size)


        # Transformer Encoder Layers
        encoder_layer = nn.TransformerEncoderLayer(d_model=hidden_size, nhead=num_heads, batch_first=True)
        self.transformer_encoder = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)

        # Final linear layer to output a single score
        self.fc = nn.Linear(hidden_size, 1)

    def forward(self, x, channel_indices, conditioning=None):
        # x is expected to be of shape (batch_size, sequence_length) - the grid sequence
        # channel_indices is expected to be of shape (batch_size, sequence_length) indicating the channel for each token
        # conditioning is expected to be of shape (batch_size, condition_size) or None

        # Embed tokens
        embedded_tokens = self.embedding(x) # shape: (batch_size, sequence_length, hidden_size)

        # Add positional embedding
        positions = torch.arange(0, x.size(1)).unsqueeze(0).repeat(x.size(0), 1).to(x.device)
        embedded_positions = self.positional_embedding(positions) # shape: (batch_size, sequence_length, hidden_size)

        # Embed channel information
        embedded_channels = self.channel_embedding(channel_indices) # shape: (batch_size, sequence_length, hidden_size)

        # Combine embeddings (summing is a common approach)
        embedded = embedded_tokens + embedded_positions + embedded_channels # shape: (batch_size, sequence_length, hidden_size)


        # Process conditioning and combine with embedded sequence (simple addition to the first token)
        if conditioning is not None:
            processed_condition = self.condition_linear(conditioning).unsqueeze(1) # shape: (batch_size, 1, hidden_size)
            # Simple approach: add to the first token's embedding
            embedded[:, 0, :] = embedded[:, 0, :] + processed_condition[:, 0, :]


        # Pass through Transformer Encoder
        output_sequence = self.transformer_encoder(embedded) # shape: (batch_size, sequence_length, hidden_size)

        # Pool the output sequence to get a single representation per sequence
        pooled_output = torch.mean(output_sequence, dim=1) # shape: (batch_size, hidden_size)

        # Pass through the final linear layer
        score = self.fc(pooled_output) # shape: (batch_size, 1)

        # Apply sigmoid activation to get a probability score
        probability = torch.sigmoid(score)

        return probability

# Helper function for Rotary Positional Embedding (RoPE)
# This is a simplified version; a full RoPE implementation might be more complex
def rotate_half(x):
    x1, x2 = x.chunk(2, dim=-1)
    return torch.cat((-x2, x1), dim=-1)

def apply_rope(x, freqs):
    # x: (batch_size, sequence_length, hidden_size)
    # freqs: (sequence_length, hidden_size)
    # Apply RoPE to the last dimension
    return (x * freqs.cos()) + (rotate_half(x) * freqs.sin())

class TransformerGenerator(nn.Module):
    def __init__(self, vocab_size, hidden_size, num_layers, num_heads, max_seq_length, noise_dim=100, condition_size=0):
        super(TransformerGenerator, self).__init__()
        self.vocab_size = vocab_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.num_heads = num_heads
        self.max_seq_length = max_seq_length
        self.noise_dim = noise_dim
        self.condition_size = condition_size

        # Embedding layer
        self.embedding = nn.Embedding(vocab_size, hidden_size)

        # Positional Encoding (Simplified RoPE-like)
        # Generate frequencies for RoPE
        freqs = 1.0 / (10000 ** (torch.arange(0, hidden_size, 2).float() / hidden_size))
        self.register_buffer('freqs', freqs.repeat_interleave(2).unsqueeze(0)) # shape: (1, hidden_size)


        # Linear layer to process noise and conditioning
        input_linear_size = self.noise_dim + self.condition_size
        self.input_linear = nn.Linear(input_linear_size, hidden_size) # Project to hidden_size


        # Transformer Decoder Layers
        # Using PyTorch's TransformerDecoderLayer and TransformerDecoder
        decoder_layer = nn.TransformerDecoderLayer(d_model=hidden_size, nhead=num_heads, batch_first=True)
        self.transformer_decoder = nn.TransformerDecoder(decoder_layer, num_layers=num_layers)

        # Final linear layer to output vocabulary scores
        self.fc = nn.Linear(hidden_size, vocab_size)

    def forward(self, x, noise, conditioning=None):
        # x is expected to be of shape (batch_size, sequence_length) - the input grid sequence
        # noise is expected to be of shape (batch_size, noise_dim)
        # conditioning is expected to be of shape (batch_size, condition_size) or None

        # Embed the input sequence
        embedded = self.embedding(x) # shape: (batch_size, sequence_length, hidden_size)

        # Apply Positional Encoding (RoPE)
        # Need to expand freqs to match sequence length
        freqs = self.freqs[:, :embedded.size(1), :] # shape: (1, sequence_length, hidden_size)
        embedded = apply_rope(embedded, freqs)

        # Process noise and conditioning and potentially use it as initial state or add to embeddings
        # For a Transformer Decoder, we often use a learned query token or cross-attention.
        # A simple way to incorporate noise/conditioning is to add it to the input embeddings
        # after processing it through a linear layer.
        processed_combined_input = self.input_linear(noise if conditioning is None else torch.cat((noise, conditioning), dim=1)) # shape: (batch_size, hidden_size)
        # Expand and add to embeddings (requires careful broadcasting or unsqueezing)
        # Let's try adding to the first token's embedding as a simple approach for now
        # embedded[:, 0, :] = embedded[:, 0, :] + processed_combined_input


        # For a standard Transformer Decoder, we often need a memory or target sequence
        # In an autoregressive setup, the input 'x' serves as both the target and part of the memory
        # We need a causal mask to prevent attending to future tokens in the output sequence portion
        # Assuming x contains [start_token, input_grid, separator_token, output_grid]
        # The causal mask should apply from the separator_token onwards.
        # This requires a more complex mask than a standard causal mask.
        # For simplicity in this initial structure, let's assume a standard causal mask for now.
        # A proper implementation for ARC would need a custom mask.
        seq_len = x.size(1)
        causal_mask = torch.triu(torch.ones(seq_len, seq_len) * float('-inf'), diagonal=1).to(x.device)


        # Pass through Transformer Decoder
        # TransformerDecoder expects target (embedded), memory (can be None if self-attention only)
        # and masks.
        # For autoregressive generation from an input sequence, the input sequence itself
        # acts as the target.
        # We might need a separate "memory" if we have an encoder part, but for a decoder-only Generator,
        # the input sequence is the primary information.

        # Let's use the embedded sequence as the target for the decoder
        output_sequence = self.transformer_decoder(embedded, memory=None, tgt_mask=causal_mask) # shape: (batch_size, sequence_length, hidden_size)


        # Final linear layer
        logits = self.fc(output_sequence) # shape: (batch_size, sequence_length, vocab_size)

        # Softmax to get probabilities (often done in the loss function for numerical stability)
        # probabilities = F.softmax(logits, dim=-1)

        return logits # Return logits for use with CrossEntropyLoss

import torch

class Generator(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, num_layers=3, noise_dim=100, condition_size=0):
        super(Generator, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.noise_dim = noise_dim
        self.condition_size = condition_size

        # Embedding layer for the input sequence (grid values and special tokens)
        self.embedding = nn.Embedding(input_size, hidden_size)

        # Linear layer to process the noise vector and potentially conditioning information
        # The input size to this layer will depend on how noise and conditioning are combined
        # A simple approach is to concatenate noise and conditioning and then pass through a linear layer
        input_linear_size = self.noise_dim + self.condition_size
        self.input_linear = nn.Linear(input_linear_size, hidden_size)


        # Stacked Bidirectional LSTM layer(s)
        # Input size is hidden_size (from embedding)
        # Output size of BiLSTM is 2 * hidden_size because it's bidirectional
        self.lstm = nn.LSTM(hidden_size, hidden_size, num_layers, batch_first=True, bidirectional=True)


        # Linear layer to output the sequence
        # The input size to the linear layer will be 2 * hidden_size due to BiLSTM
        self.fc = nn.Linear(hidden_size * 2, output_size) # output_size would be the vocabulary size again

    def forward(self, x, noise, conditioning=None):
        # x is expected to be of shape (batch_size, sequence_length) - the input grid sequence
        # noise is expected to be of shape (batch_size, noise_dim)
        # conditioning is expected to be of shape (batch_size, condition_size) or None

        # Embed the input sequence
        embedded = self.embedding(x) # shape: (batch_size, sequence_length, hidden_size)

        # Process noise and conditioning
        if conditioning is not None:
            combined_input = torch.cat((noise, conditioning), dim=1)
        else:
            combined_input = noise

        # Pass combined input through linear layer - this output could be used to initialize LSTM state
        # or be incorporated differently. For now, let's just process it.
        processed_combined_input = self.input_linear(combined_input) # shape: (batch_size, hidden_size)

        # Pass embedded sequence through LSTM
        # We can potentially use processed_combined_input to initialize the hidden state of the LSTM
        # h0 = processed_combined_input.unsqueeze(0).repeat(self.num_layers * 2, 1, 1) # *2 for bidirectional
        # c0 = torch.zeros(self.num_layers * 2, x.size(0), self.hidden_size).to(x.device) # *2 for bidirectional
        # out, _ = self.lstm(embedded, (h0, c0))

        # For simplicity, let's just pass the embedded sequence through LSTM
        out, _ = self.lstm(embedded) # shape: (batch_size, sequence_length, 2 * hidden_size)


        # Pass through the linear layer to get output predictions
        # We apply the linear layer to each time step
        out = self.fc(out) # shape: (batch_size, sequence_length, output_size)

        return out

class Generator(nn.Module):
    def __init__(self, input_size, hidden_size, output_size, num_layers=3, noise_dim=100, condition_size=0):
        super(Generator, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.noise_dim = noise_dim
        self.condition_size = condition_size
        self.input_size = input_size
        self.output_size = output_size

        self.embedding = nn.Embedding(input_size, hidden_size)

        input_linear_size = self.noise_dim + self.condition_size
        self.input_linear = nn.Linear(input_linear_size, hidden_size)

        self.lstm = nn.LSTM(hidden_size, hidden_size, num_layers, batch_first=True, bidirectional=True)

        self.fc = nn.Linear(hidden_size * 2, output_size)

    def forward(self, x, noise, conditioning=None):
        embedded = self.embedding(x)

        if conditioning is not None:
            combined_input = torch.cat((noise, conditioning), dim=1)
        else:
            combined_input = noise

        processed_combined_input = self.input_linear(combined_input)

        h0 = processed_combined_input.unsqueeze(0).repeat(self.num_layers * 2, 1, 1)
        c0 = torch.zeros(self.num_layers * 2, x.size(0), self.hidden_size).to(x.device)

        out, _ = self.lstm(embedded, (h0, c0))

        out = self.fc(out)

        return out

# Helper function for Rotary Positional Embedding (RoPE) - simplified
def rotate_half(x):
    x1, x2 = x.chunk(2, dim=-1)
    return torch.cat((-x2, x1), dim=-1)

def apply_rope(x, freqs):
    # x: (batch_size, sequence_length, hidden_size)
    # freqs: (sequence_length, hidden_size) - needs to be expanded to match batch_size and sequence_length
    # Apply RoPE to the last dimension
    return (x * freqs.cos()) + (rotate_half(x) * freqs.sin())

# Define the TransformerDiscriminator class
class TransformerDiscriminator(nn.Module):
    def __init__(self, vocab_size, hidden_size, num_layers, num_heads, max_seq_length, condition_size=0, num_channels=2):
        super(TransformerDiscriminator, self).__init__()
        self.vocab_size = vocab_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.num_heads = num_heads
        self.max_seq_length = max_seq_length
        self.condition_size = condition_size
        self.num_channels = num_channels # Number of channels (e.g., input and output)

        self.embedding = nn.Embedding(vocab_size, hidden_size)
        self.positional_embedding = nn.Embedding(max_seq_length, hidden_size)
        self.channel_embedding = nn.Embedding(num_channels, hidden_size)

        if self.condition_size > 0:
             self.condition_linear = nn.Linear(condition_size, hidden_size)

        encoder_layer = nn.TransformerEncoderLayer(d_model=hidden_size, nhead=num_heads, batch_first=True)
        self.transformer_encoder = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)

        self.fc = nn.Linear(hidden_size, 1)

    def forward(self, x, channel_indices, conditioning=None):
        embedded_tokens = self.embedding(x)
        positions = torch.arange(0, x.size(1)).unsqueeze(0).repeat(x.size(0), 1).to(x.device)
        embedded_positions = self.positional_embedding(positions)
        embedded_channels = self.channel_embedding(channel_indices)

        embedded = embedded_tokens + embedded_positions + embedded_channels

        if conditioning is not None:
            processed_condition = self.condition_linear(conditioning).unsqueeze(1)
            embedded[:, 0, :] = embedded[:, 0, :] + processed_condition[:, 0, :]

        output_sequence = self.transformer_encoder(embedded)
        pooled_output = torch.mean(output_sequence, dim=1)
        score = self.fc(pooled_output)
        probability = torch.sigmoid(score)

        return probability

# Define the TransformerGenerator class
class TransformerGenerator(nn.Module):
    def __init__(self, vocab_size, hidden_size, num_layers, num_heads, max_seq_length, noise_dim=100, condition_size=0):
        super(TransformerGenerator, self).__init__()
        self.vocab_size = vocab_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.num_heads = num_heads
        self.max_seq_length = max_seq_length
        self.noise_dim = noise_dim
        self.condition_size = condition_size

        self.embedding = nn.Embedding(vocab_size, hidden_size)

        freqs = 1.0 / (10000 ** (torch.arange(0, hidden_size, 2).float() / hidden_size))
        self.register_buffer('freqs', freqs.repeat_interleave(2).unsqueeze(0))

        input_linear_size = self.noise_dim + self.condition_size
        self.input_linear = nn.Linear(input_linear_size, hidden_size)

        decoder_layer = nn.TransformerDecoderLayer(d_model=hidden_size, nhead=num_heads, batch_first=True)
        self.transformer_decoder = nn.TransformerDecoder(decoder_layer, num_layers=num_layers)

        self.fc = nn.Linear(hidden_size, vocab_size)

    def forward(self, x, noise, conditioning=None):
        embedded = self.embedding(x)

        seq_len = embedded.size(1)
        freqs_expanded_for_rope = self.freqs[:, :seq_len, :].expand(x.size(0), seq_len, -1)
        embedded = apply_rope(embedded, freqs_expanded_for_rope)

        processed_combined_input = self.input_linear(noise if conditioning is None else torch.cat((noise, conditioning), dim=1))
        processed_combined_input_expanded = processed_combined_input.unsqueeze(1).expand(-1, seq_len, -1)
        embedded = embedded + processed_combined_input_expanded

        causal_mask = torch.triu(torch.ones(seq_len, seq_len) * float('-inf'), diagonal=1).to(x.device)

        output_sequence = self.transformer_decoder(embedded, memory=None, tgt_mask=causal_mask)

        logits = self.fc(output_sequence)

        return logits

# Define a function to train one epoch of the GAN
def train_gan_one_epoch(generator, discriminator, train_dataloader, discriminator_criterion, generator_criterion, generator_optimizer, discriminator_optimizer, device, noise_dim, value_to_index, separator_token):
    generator.train()
    discriminator.train()

    discriminator_loss_sum = 0.0
    generator_loss_sum = 0.0

    separator_mapped_index = value_to_index[separator_token]

    for i, real_sequences in enumerate(tqdm(train_dataloader, desc=f"Training Batch")):
        real_sequences = real_sequences.to(device)
        real_sequences_mapped = torch.zeros_like(real_sequences, dtype=torch.long).to(device)
        for original_value, mapped_index in value_to_index.items():
             real_sequences_mapped[real_sequences == original_value] = mapped_index

        batch_size = real_sequences.size(0)
        sequence_length = real_sequences.size(1)

        real_labels = torch.ones(batch_size, 1).to(device)
        fake_labels = torch.zeros(batch_size, 1).to(device)

        # Determine channel indices for real data
        channel_indices_real = torch.zeros(batch_size, sequence_length, dtype=torch.long).to(device)
        separator_locations = (real_sequences_mapped == separator_mapped_index).nonzero(as_tuple=True)
        separator_indices = -torch.ones(batch_size, dtype=torch.long).to(device)
        for b_idx, s_idx in zip(*separator_locations):
            if separator_indices[b_idx] == -1:
                separator_indices[b_idx] = s_idx
        for b_idx in range(batch_size):
             if separator_indices[b_idx] != -1:
                channel_indices_real[b_idx, separator_indices[b_idx] + 1:] = 1

        # Train Discriminator
        discriminator_optimizer.zero_grad()

        # Real data
        real_output = discriminator(real_sequences_mapped, channel_indices_real, conditioning=None)
        discriminator_real_loss = discriminator_criterion(real_output, real_labels)

        # Fake data
        noise = torch.randn(batch_size, noise_dim).to(device)
        fake_sequences_logits = generator(real_sequences_mapped, noise, conditioning=None)
        fake_sequences_mapped = torch.argmax(fake_sequences_logits, dim=-1)

        channel_indices_fake = torch.zeros(batch_size, sequence_length, dtype=torch.long).to(device)
        separator_locations_fake = (fake_sequences_mapped == separator_mapped_index).nonzero(as_tuple=True)
        separator_indices_fake = -torch.ones(batch_size, dtype=torch.long).to(device)
        for b_idx, s_idx in zip(*separator_locations_fake):
            if separator_indices_fake[b_idx] == -1:
                separator_indices_fake[b_idx] = s_idx
        for b_idx in range(batch_size):
             if separator_indices_fake[b_idx] != -1:
                channel_indices_fake[b_idx, separator_indices_fake[b_idx] + 1:] = 1


        fake_output = discriminator(fake_sequences_mapped.detach(), channel_indices_fake, conditioning=None)
        discriminator_fake_loss = discriminator_criterion(fake_output, fake_labels)

        discriminator_loss = discriminator_real_loss + discriminator_fake_loss
        discriminator_loss.backward()
        discriminator_optimizer.step()

        # Train Generator
        generator_optimizer.zero_grad()

        noise = torch.randn(batch_size, noise_dim).to(device)
        fake_sequences_logits = generator(real_sequences_mapped, noise, conditioning=None)
        fake_sequences_mapped = torch.argmax(fake_sequences_logits, dim=-1)

        channel_indices_fake = torch.zeros(batch_size, sequence_length, dtype=torch.long).to(device)
        separator_locations_fake = (fake_sequences_mapped == separator_mapped_index).nonzero(as_tuple=True)
        separator_indices_fake = -torch.ones(batch_size, dtype=torch.long).to(device)
        for b_idx, s_idx in zip(*separator_locations_fake):
            if separator_indices_fake[b_idx] == -1:
                separator_indices_fake[b_idx] = s_idx
        for b_idx in range(batch_size):
             if separator_indices_fake[b_idx] != -1:
                channel_indices_fake[b_idx, separator_indices_fake[b_idx] + 1:] = 1


        output_from_discriminator_on_fake = discriminator(fake_sequences_mapped, channel_indices_fake, conditioning=None)
        generator_loss = generator_criterion(output_from_discriminator_on_fake, real_labels)

        generator_loss.backward()
        generator_optimizer.step()

        discriminator_loss_sum += discriminator_loss.item()
        generator_loss_sum += generator_loss.item()

    avg_discriminator_loss_epoch = discriminator_loss_sum / len(train_dataloader)
    avg_generator_loss_epoch = generator_loss_sum / len(train_dataloader)

    return avg_discriminator_loss_epoch, avg_generator_loss_epoch

        # but can help monitor training stability.
