class BaseDataLoader:
    """Base class for creating a tf.data.Dataset pipeline.
    Handles common pre-processing steps like padding/cropping to a fixed size,
    channel addition, and batching.
    """
    def __init__(self, config):
        self.config = config

    def _pad_and_crop_grid_tf(self, grid_tensor):
        """Pads or crops a grid tensor to a fixed square size (config.grid_size)."""
        height, width = tf.shape(grid_tensor)[0], tf.shape(grid_tensor)[1]

        # Calculate padding amounts
        pad_height = tf.maximum(0, self.config.grid_size - height)
        pad_width = tf.maximum(0, self.config.grid_size - width)

        # Pad only if needed
        padded_grid = tf.pad(
            grid_tensor,
            [[0, pad_height], [0, pad_width]],
            constant_values=0 # Use 0 for padding as it's a valid color and common background
        )

        # Crop if the original grid (or padded grid) is larger than the target size
        # This will ensure the grid is exactly self.config.grid_size x self.config.grid_size
        cropped_grid = padded_grid[:self.config.grid_size, :self.config.grid_size]

        return tf.cast(cropped_grid, dtype=tf.int32)

    def _get_data_generator(self):
        """Placeholder for a generator function, implemented by subclasses."""
        raise NotImplementedError("Subclasses must implement _get_data_generator.")

    def load_dataset(self):
        """Returns a batched, pre-processed tf.data.Dataset from the generator."""
        logging.info("Starting dataset loading and preprocessing pipeline...")
        ds = tf.data.Dataset.from_generator(
            self._get_data_generator,
            output_signature=tf.TensorSpec(shape=(None, None), dtype=tf.int32) # Grids of varying (H,W)
        )

        # Filter out empty or invalid grids early
        ds = ds.filter(lambda x: tf.rank(x) == 2 and tf.shape(x)[0] > 0 and tf.shape(x)[1] > 0)

        # Apply padding/cropping to a uniform size (e.g., 30x30)
        ds = ds.map(self._pad_and_crop_grid_tf, num_parallel_calls=tf.data.AUTOTUNE)

        # Add a channel dimension (e.g., (H, W) -> (H, W, 1)) and cast to float32
        # VAE Encoder expects (H,W,1) float32 input.
        ds = ds.map(lambda x: tf.expand_dims(tf.cast(x, tf.float32), axis=-1), num_parallel_calls=tf.data.AUTOTUNE)

        # After _pad_and_crop_grid_tf, the grids are guaranteed to be self.config.grid_size x self.config.grid_size.
        # After expand_dims, they are self.config.grid_size x self.config.grid_size x 1.
        # No need for an additional filter based on shape.

        ds = ds.shuffle(buffer_size=5000)
        ds = ds.batch(self.config.batch_size, drop_remainder=True) # drop_remainder ensures fixed batch shape
        ds = ds.prefetch(tf.data.AUTOTUNE)
        logging.info("Dataset pipeline created and ready.")
        return ds

class ARCDataLoader(BaseDataLoader):
    """Data loader for the official ARC training challenges."""
    def _get_data_generator(self):
        """Loads grids from the official ARC training challenges JSON file."""
        try:
            with open(self.config.training_challenges_path, 'r') as f:
                full_data = json.load(f)
        except FileNotFoundError:
            logging.error(f"ARC training challenges file not found at: {self.config.training_challenges_path}")
            sys.exit(1) # Exit if essential file is missing
        except json.JSONDecodeError as e:
            logging.error(f"Error decoding JSON from {self.config.training_challenges_path}: {e}")
            sys.exit(1)

        logging.info(f"Loading ARC training challenges from {self.config.training_challenges_path}")
        for task_id, task_data in full_data.items():
            for example in task_data.get('train', []):
                if 'output' in example and example['output'] is not None:
                    output_grid = np.array(example['output'], dtype=np.int32)
                    if output_grid.size > 0: # Ensure grid is not empty after conversion
                        yield output_grid

class SyntheticDataLoader(BaseDataLoader):
    """Data loader for GAN-generated synthetic grids."""
    def _get_data_generator(self):
        """Loads grids from multiple synthetic dataset JSON files."""
        for path in self.config.synthetic_data_paths:
            try:
                with open(path, 'r') as f:
                    data = json.load(f)
                    logging.info(f"Loading synthetic data from {path}")
                    for item in data:
                        grid = None
                        if isinstance(item, list) and all(isinstance(sublist, list) for sublist in item):
                            grid = np.array(item, dtype=np.int32)
                        elif isinstance(item, list) and all(isinstance(val, (int, float)) for val in item):
                            grid_size_sq = len(item)
                            grid_size = int(np.sqrt(grid_size_sq))
                            if grid_size * grid_size == grid_size_sq:
                                grid = np.array(item, dtype=np.int32).reshape(grid_size, grid_size)
                            else:
                                logging.warning(f"Skipping malformed flattened grid in {path} (non-square length: {grid_size_sq}).")
                        else:
                            logging.warning(f"Skipping malformed data item in {path}: {type(item)}.")

                        if grid is not None and grid.size > 0:
                            yield grid

            except (FileNotFoundError, json.JSONDecodeError) as e:
                logging.warning(f"Skipping file {path} due to error: {e}")
            except Exception as e:
                logging.error(f"An unexpected error occurred while loading {path}: {e}")

class Sampling(layers.Layer):
    """Uses (z_mean, z_log_var) to sample z, the vector encoding a digit."""
    def call(self, inputs):
        z_mean, z_log_var = inputs
        batch = tf.shape(z_mean)[0]
        dim = tf.shape(z_mean)[1]
        epsilon = tf.random.normal(shape=(batch, dim))
        return z_mean + tf.exp(0.5 * z_log_var) * epsilon

class Encoder(Model):
    def __init__(self, latent_dim, grid_size, name='encoder', **kwargs):
        super(Encoder, self).__init__(name=name, **kwargs)
        self.latent_dim = latent_dim
        self.grid_size = grid_size
        self.conv1 = layers.Conv2D(32, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn1 = layers.BatchNormalization()
        self.conv2 = layers.Conv2D(64, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn2 = layers.BatchNormalization()
        self.flatten = layers.Flatten()
        self.z_mean_dense = layers.Dense(latent_dim, name='z_mean')
        self.z_log_var_dense = layers.Dense(latent_dim, name='z_log_var')

    def call(self, inputs, training=None):
        x = self.conv1(inputs)
        x = self.bn1(x, training=training)
        x = self.conv2(x)
        x = self.bn2(x, training=training)
        x = self.flatten(x)
        z_mean = self.z_mean_dense(x)
        z_log_var = self.z_log_var_dense(x)
        return z_mean, z_log_var

class Decoder(layers.Layer):
    def __init__(self, latent_dim, num_colors, grid_size, name='decoder', **kwargs):
        super(Decoder, self).__init__(name=name, **kwargs)
        self.num_colors = num_colors
        self.latent_dim = latent_dim
        self.grid_size = grid_size # Store for resizing output

        # Calculate initial dimensions for deconvolution based on fixed grid_size
        # For grid_size=30, two stride-2 convs (padding='same') yield 8x8 spatial dims.
        # ceil(ceil(grid_size / 2) / 2)
        initial_h_float = tf.math.ceil(tf.cast(grid_size, tf.float32) / 2.0)
        initial_h_float = tf.math.ceil(initial_h_float / 2.0)
        self.initial_h = tf.cast(initial_h_float, tf.int32).numpy()
        self.initial_w = self.initial_h # Assuming square grids

        self.dense = layers.Dense(self.initial_h * self.initial_w * 16, activation='relu')
        self.reshape = layers.Reshape((self.initial_h, self.initial_w, 16))

        self.deconv1 = layers.Conv2DTranspose(64, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn1 = layers.BatchNormalization()
        self.deconv2 = layers.Conv2DTranspose(32, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn2 = layers.BatchNormalization()

        self.output_conv = layers.Conv2DTranspose(num_colors, 3, activation='softmax', padding='same', kernel_initializer=GlorotNormal())

    def call(self, inputs):
        x = self.dense(inputs)
        x = self.reshape(x)
        x = self.deconv1(x)
        x = self.bn1(x)
        x = self.deconv2(x)
        x = self.bn2(x)
        output_logits = self.output_conv(x)

        # CRITICAL FIX: Ensure output matches target grid_size
        # The deconv sequence 8->16->32 results in 32x32, but target is grid_size (e.g., 30).
        # Resize to explicitly match the config.grid_size
        if output_logits.shape[1] != self.grid_size or output_logits.shape[2] != self.grid_size:
            output_logits = tf.image.resize(
                output_logits,
                [self.grid_size, self.grid_size],
                method=tf.image.ResizeMethod.NEAREST_NEIGHBOR # Nearest neighbor for categorical predictions
            )
        return output_logits

class VAE(Model):
    def __init__(self, latent_dim, num_colors, grid_size, name="vae", **kwargs):
        super(VAE, self).__init__(name=name, **kwargs)
        self.encoder = Encoder(latent_dim=latent_dim, grid_size=grid_size)
        self.sampling = Sampling()
        self.decoder = Decoder(latent_dim=latent_dim, num_colors=num_colors, grid_size=grid_size)
        self.total_loss_tracker = tf.keras.metrics.Mean(name="total_loss")
        self.reconstruction_loss_tracker = tf.keras.metrics.Mean(name="reconstruction_loss")
        self.kl_loss_tracker = tf.keras.metrics.Mean(name="kl_loss")

    def call(self, inputs, training=None):
        # inputs are expected to be (batch, H, W, 1) float32
        z_mean, z_log_var = self.encoder(inputs, training=training) # Pass training to encoder
        z = self.sampling([z_mean, z_log_var])
        reconstruction = self.decoder(z)
        return reconstruction

    @property
    def metrics(self):
        return [
            self.total_loss_tracker,
            self.reconstruction_loss_tracker,
            self.kl_loss_tracker,
        ]

    def train_step(self, data):
        # `data` is the input grid, which is also the target for reconstruction.
        # Expected shape: (batch, H, W, 1), float32

        with tf.GradientTape() as tape:
            z_mean, z_log_var = self.encoder(data, training=True)
            z = self.sampling([z_mean, z_log_var])
            reconstruction = self.decoder(z)

            # Reconstruction loss: Compare `reconstruction` (softmax probabilities)
            # with `data` (integer labels) using SparseCategoricalCrossentropy.
            # Flatten to calculate loss efficiently
            reconstruction_flattened = tf.reshape(reconstruction, [-1, self.decoder.num_colors])
            # Target `data` needs to be int32 and flattened
            data_flattened_int = tf.reshape(tf.cast(data, tf.int32), [-1])

            reconstruction_loss = tf.keras.losses.sparse_categorical_crossentropy(
                data_flattened_int, reconstruction_flattened, from_logits=False
            )
            reconstruction_loss = tf.reduce_mean(reconstruction_loss) # Mean over all pixels in batch

            # KL divergence loss
            kl_loss = -0.5 * (1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var))
            kl_loss = tf.reduce_mean(tf.reduce_sum(kl_loss, axis=1)) # Sum over latent_dim, then mean over batch

            total_loss = reconstruction_loss + kl_loss

        grads = tape.gradient(total_loss, self.trainable_weights)
        self.optimizer.apply_gradients(zip(grads, self.trainable_weights))

        self.total_loss_tracker.update_state(total_loss)
        self.reconstruction_loss_tracker.update_state(reconstruction_loss)
        self.kl_loss_tracker.update_state(kl_loss)
        return {
            "loss": self.total_loss_tracker.result(),
            "reconstruction_loss": self.reconstruction_loss_tracker.result(),
            "kl_loss": self.kl_loss_tracker.result(),
        }

    def test_step(self, data):
        # `data` is the input grid, which is also the target for reconstruction.
        # Expected shape: (batch, H, W, 1), float32

        z_mean, z_log_var = self.encoder(data, training=False)
        z = self.sampling([z_mean, z_log_var])
        reconstruction = self.decoder(z)

        reconstruction_flattened = tf.reshape(reconstruction, [-1, self.decoder.num_colors])
        data_flattened_int = tf.reshape(tf.cast(data, tf.int32), [-1])

        reconstruction_loss = tf.keras.losses.sparse_categorical_crossentropy(
            data_flattened_int, reconstruction_flattened, from_logits=False
        )
        reconstruction_loss = tf.reduce_mean(reconstruction_loss)

        kl_loss = -0.5 * (1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var))
        kl_loss = tf.reduce_mean(tf.reduce_sum(kl_loss, axis=1))

        total_loss = reconstruction_loss + kl_loss

        self.total_loss_tracker.update_state(total_loss)
        self.reconstruction_loss_tracker.update_state(reconstruction_loss)
        self.kl_loss_tracker.update_state(kl_loss)
        return {
            "loss": self.total_loss_tracker.result(),
            "reconstruction_loss": self.reconstruction_loss_tracker.result(),
            "kl_loss": self.kl_loss_tracker.result(),
        }

class ModelTrainer:
    def __init__(self, config, model, dataset):
        self.config = config
        self.model = model
        self.dataset = dataset
        # FIX: Use learning_rate from config
        self.model.compile(optimizer=Adam(learning_rate=self.config.learning_rate))

        # FIX: Define checkpoint path and ensure it ends with .weights.h5
        os.makedirs(self.config.vae_checkpoints_dir, exist_ok=True)
        # --- MODIFICATION HERE ---
        self.checkpoint_filepath = os.path.join(self.config.vae_checkpoints_dir, 'vae_weights_best.weights.h5')
        logging.info(f"Model checkpoints will be saved to: {self.checkpoint_filepath}")


    def train(self):
        callbacks = [
            TensorBoard(log_dir='./logs'),
            # FIX: Use ModelCheckpoint for robust saving of the best model
            ModelCheckpoint(
                filepath=self.checkpoint_filepath,
                save_weights_only=True,
                monitor='reconstruction_loss', # Monitor reconstruction loss to find best model
                mode='min',
                save_best_only=True,
                verbose=1
            ),
            EarlyStopping(monitor='reconstruction_loss', patience=5, verbose=1) # Adjust patience if needed
        ]
        logging.info(f"Starting VAE training for {self.config.epochs} epochs...")
        self.model.fit(
            self.dataset,
            epochs=self.config.epochs,
            callbacks=callbacks
        )
        logging.info("VAE training complete.")

        # Load the best weights back into the model after training
        try:
            self.model.load_weights(self.checkpoint_filepath)
            logging.info(f"Loaded best weights from {self.checkpoint_filepath}")
        except tf.errors.NotFoundError:
            logging.warning(f"No best weights found at {self.checkpoint_filepath}. Using last epoch's weights.")


class PlausibilityValidator:
    """Uses the trained VAE to score the plausibility of an ARC grid."""
    def __init__(self, vae_model, grid_size, num_colors):
        self.vae_model = vae_model
        self.grid_size = grid_size
        self.num_colors = num_colors
        self.loss_fn = SparseCategoricalCrossentropy(from_logits=False)

    def get_plausibility_score(self, grid):
        """Calculates a plausibility score based on VAE's reconstruction error."""
        # Ensure input grid is int32 and add batch and channel dimensions for VAE input
        # It's crucial for the grid to be preprocessed to the VAE's expected input size
        # and format (H, W, 1) float32.
        if tf.rank(grid) == 2:
            grid = tf.expand_dims(grid, axis=-1) # Add channel dim
        elif tf.rank(grid) == 3 and grid.shape[-1] != 1:
            grid = grid[..., :1] # Keep only first channel if multiple present

        # Ensure correct size (will be handled by the DataLoader in general, but good for direct input)
        grid_resized = tf.image.resize_with_crop_or_pad(
            tf.cast(grid, tf.float32),
            target_height=self.grid_size,
            target_width=self.grid_size
        )

        grid_tensor_input = tf.expand_dims(grid_resized, axis=0) # Add batch dimension (1, H, W, 1)

        reconstruction = self.vae_model(grid_tensor_input, training=False)

        # Prepare the target grid for loss calculation
        # Cast to int32, and ensure it's (1, H, W) for SparseCategoricalCrossentropy
        grid_tensor_loss_target = tf.cast(tf.squeeze(grid_resized, axis=-1), dtype=tf.int32) # Remove channel for target
        grid_tensor_loss_target = tf.expand_dims(grid_tensor_loss_target, axis=0) # Add batch dimension (1, H, W)

        reconstruction_loss = self.loss_fn(grid_tensor_loss_target, reconstruction)
        # Higher score for more plausible grids
        return tf.exp(-reconstruction_loss).numpy()

class RLAgent:
    """
    A Reinforcement Learning agent using a Deep Q-Network (DQN) to generate
    plausible ARC grids based on feedback from a trained VAE.
    """
    def __init__(self, vae_model, plausibility_validator, grid_size, num_colors, learning_rate=0.001,
                 discount_factor=0.99, exploration_rate=1.0, min_exploration_rate=0.01,
                 exploration_decay_rate=0.995):
        self.vae_model = vae_model # Keep VAE reference if needed for other VAE ops
        self.plausibility_validator = plausibility_validator # FIX: Use the validator instance
        self.grid_size = grid_size
        self.num_colors = num_colors
        self.action_space_size = grid_size * grid_size * num_colors # Each cell (x,y) can be any color (c)
        self.learning_rate = learning_rate
        self.discount_factor = discount_factor
        self.exploration_rate = exploration_rate
        self.min_exploration_rate = min_exploration_rate
        self.exploration_decay_rate = exploration_decay_rate
        self.q_network = self._build_q_network()
        self.optimizer = Adam(learning_rate=self.learning_rate)

    def _build_q_network(self):
        inputs = layers.Input(shape=(self.grid_size, self.grid_size, 1), dtype=tf.float32)
        x = layers.Conv2D(32, 3, activation='relu', padding='same')(inputs)
        x = layers.Flatten()(x)
        x = layers.Dense(128, activation='relu')(x)
        outputs = layers.Dense(self.action_space_size)(x) # Output Q-values for all possible actions
        return Model(inputs=inputs, outputs=outputs)

    def choose_action(self, state_grid):
        if np.random.uniform(0, 1) < self.exploration_rate:
            action_index = np.random.randint(self.action_space_size)
        else:
            # Prepare state_grid for Q-network: (1, H, W, 1) float32
            state_tensor = tf.cast(tf.expand_dims(tf.expand_dims(state_grid, 0), -1), dtype=tf.float32)
            q_values = self.q_network(state_tensor, training=False)
            action_index = tf.argmax(q_values[0]).numpy()
        return self._map_index_to_action(action_index)

    def _map_index_to_action(self, index):
        """Maps a linear action index to (x, y, color)."""
        color = index % self.num_colors
        index //= self.num_colors # Remaining index for (x,y)
        y = index % self.grid_size
        x = index // self.grid_size
        return (x, y, color)

    def _map_action_to_index(self, x, y, color):
        """Maps (x, y, color) to a linear action index."""
        return x * self.grid_size * self.num_colors + y * self.num_colors + color

    @tf.function # For performance
    def train_step(self, state, action, reward, next_state, done):
        state_tensor = tf.cast(tf.expand_dims(tf.expand_dims(state, 0), -1), dtype=tf.float32)
        next_state_tensor = tf.cast(tf.expand_dims(tf.expand_dims(next_state, 0), -1), dtype=tf.float32)

        with tf.GradientTape() as tape:
            q_values = self.q_network(state_tensor)

            if done:
                target_q_value = reward
            else:
                next_q_values = self.q_network(next_state_tensor)
                max_next_q_value = tf.reduce_max(next_q_values)
                target_q_value = reward + self.discount_factor * max_next_q_value

            action_index = self._map_action_to_index(*action)
            action_mask = tf.one_hot(action_index, self.action_space_size, dtype=tf.float32)

            q_value_for_action = tf.reduce_sum(q_values * action_mask, axis=1) # Get Q-value for chosen action

            loss = tf.keras.losses.MSE(tf.expand_dims(target_q_value, 0), q_value_for_action)

        grads = tape.gradient(loss, self.q_network.trainable_variables)
        self.optimizer.apply_gradients(zip(grads, self.q_network.trainable_variables))

    def get_vae_reward(self, grid):
        # FIX: Use the actual plausibility_validator instance
        plausibility_score = self.plausibility_validator.get_plausibility_score(grid)
        # FIX: Reward should be high for plausible grids, so use score directly
        return float(plausibility_score)

    def decay_exploration_rate(self):
        self.exploration_rate = max(self.min_exploration_rate, self.exploration_rate * self.exploration_decay_rate)

    def generate_grid_with_rl(self, max_steps=1000):
        logging.info("RL Agent: Starting grid generation...")
        current_grid = np.zeros((self.grid_size, self.grid_size), dtype=np.int32)
        # Smallest possible grid to make it plausible, e.g., one filled cell
        current_grid[0,0] = 1 # Start with a minimal plausible element

        for step in range(max_steps):
            action = self.choose_action(current_grid)
            x, y, color = action

            next_grid = current_grid.copy()
            next_grid[x, y] = color # Apply the action

            # Compute reward for the new state (next_grid)
            reward = self.get_vae_reward(next_grid)

            done = step == max_steps - 1 # Episode ends after max_steps

            self.train_step(current_grid, action, reward, next_grid, done) # Train the Q-network

            current_grid = next_grid # Move to the next state
            self.decay_exploration_rate() # Decay exploration

            if step % 100 == 0:
                logging.info(f"RL Agent: Step {step}/{max_steps}, Reward: {reward:.4f}, Exploration: {self.exploration_rate:.4f}")

        logging.info("RL Agent: Grid generation complete.")
        return current_grid

def main():
    config = Config()
    tf.config.run_functions_eagerly(config.run_eagerly) # Set eager execution mode

    # Set random seeds for reproducibility
    tf.random.set_seed(42)
    np.random.seed(42)

    logging.info("Initializing DataLoaders...")
    arc_data_loader = ARCDataLoader(config)
    synthetic_data_loader = SyntheticDataLoader(config)

    arc_ds = arc_data_loader.load_dataset()
    synthetic_ds = synthetic_data_loader.load_dataset()

    # Combine datasets
    combined_ds = arc_ds.concatenate(synthetic_ds)
    # Shuffle again after concatenation to mix ARC and synthetic data
    combined_ds = combined_ds.shuffle(buffer_size=5000)

    # Optional: Verify the shapes after preprocessing and batching
    logging.info("Verifying dataset output shape after preprocessing and batching:")
    for element in combined_ds.take(1):
        logging.info(f"  Shape of first batch: {element.shape}")
        # Expected: (config.batch_size, config.grid_size, config.grid_size, 1)
        break
    logging.info("-" * 50)

    # Model setup and training
    logging.info("Initializing VAE model...")
    vae = VAE(config.latent_dim, config.num_colors, config.grid_size)
    # Explicitly build the VAE model with the expected input shape
    vae.build(input_shape=(None, config.grid_size, config.grid_size, 1))

    # --- MODIFICATION HERE ---
    # Perform a dummy forward pass to ensure all layer weights are built
    dummy_input = tf.zeros((1, config.grid_size, config.grid_size, 1), dtype=tf.float32)
    _ = vae(dummy_input)

    vae.summary(line_length=120) # Print model summary
    logging.info(f"VAE model built. Encoder input shape: (None, {config.grid_size}, {config.grid_size}, 1)")

    logging.info("Initializing ModelTrainer and starting VAE training...")
    trainer = ModelTrainer(config, vae, combined_ds)
    trainer.train()
    logging.info("Main execution complete.")

    # 5. Plausibility Validator Route
    logging.info("\nDemonstrating Plausibility Validator...")
    # Initialize validator with the trained VAE
    validator = PlausibilityValidator(vae, config.grid_size, config.num_colors)

    # Example 1: A simple, relatively plausible grid (a square)
    plausible_grid_np = np.zeros((config.grid_size, config.grid_size), dtype=np.int32)
    plausible_grid_np[5:10, 5:10] = 1 # A 5x5 square of color 1
    plausibility_score_1 = validator.get_plausibility_score(plausible_grid_np)
    logging.info(f"Plausibility score for a simple square grid: {plausibility_score_1:.4f}")

    # Example 2: An implausible grid (random noise)
    implausible_grid_np = np.random.randint(0, config.num_colors, size=(config.grid_size, config.grid_size), dtype=np.int32)
    plausibility_score_2 = validator.get_plausibility_score(implausible_grid_np)
    logging.info(f"Plausibility score for a random noise grid: {plausibility_score_2:.4f}")

    # 6. RL Agent Route
    logging.info("\nDemonstrating RL Agent grid generation...")
    # Initialize RL agent with the trained VAE and the PlausibilityValidator
    rl_agent = RLAgent(vae_model=vae,
                       plausibility_validator=validator, # Pass the validator instance
                       grid_size=config.grid_size,
                       num_colors=config.num_colors,
                       learning_rate=config.learning_rate)
    generated_grid = rl_agent.generate_grid_with_rl(max_steps=500) # Reduced steps for demonstration
    logging.info("\nGenerated a grid using the RL agent:")
    # Print a small part of the grid or a simplified representation
    for row in generated_grid[:10, :10]: # Print top-left 10x10 for brevity
        logging.info(' '.join(map(str, row)))
    final_plausibility = validator.get_plausibility_score(generated_grid)
    logging.info(f"Final plausibility score of RL-generated grid: {final_plausibility:.4f}")
    logging.info("Full main execution complete.")


if __name__ == "__main__":
    main()
