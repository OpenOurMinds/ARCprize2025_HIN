class BaseDataLoader:
    """Base class for creating a tf.data.Dataset pipeline.
    Handles common pre-processing steps like padding, one-hot encoding, and batching.
    """
    def __init__(self, config):
        self.config = config

    def preprocess_arc_grid(self, grid_tensor):
        """
        Resizes, pads, or crops an input grid tensor to a fixed target_size,
        ensures it's float32, and explicitly sets the channel dimension to 1.
        This method is now part of BaseDataLoader and uses self.target_size.
        """
        # 1. Cast to float32
        grid_tensor = tf.cast(grid_tensor, dtype=tf.float32)

        # 2. Ensure it has a channel dimension and that dimension is 1.
        current_rank = tf.rank(grid_tensor)

        if current_rank == 2:
            # Input is (H, W) -> expand to (H, W, 1)
            grid_tensor = tf.expand_dims(grid_tensor, axis=-1)
        elif current_rank == 3:
            # Input is (H, W, C). Ensure C becomes 1.
            # This handles cases where C is None, or C > 1.
            grid_tensor = grid_tensor[..., :1] # Slice to get only the first channel (H, W, 1)
        else:
            tf.Assert(False, ["Unsupported grid_tensor rank (expected 2 or 3): " + tf.as_string(current_rank)])

        # Now, grid_tensor is guaranteed to be (H_orig, W_orig, 1)

        # 3. Resize/Pad/Crop to the target_size
        grid_tensor = tf.image.resize_with_crop_or_pad(
            grid_tensor,
            target_height=self.target_size, # Use self.target_size
            target_width=self.target_size    # Use self.target_size
        )
        return grid_tensor

    def _pad_grid_tf(self, grid_tensor):
        """Pads or crops a grid tensor to a fixed square size (config.grid_size)."""
        height, width = tf.shape(grid_tensor)[0], tf.shape(grid_tensor)[1]

        # Calculate padding amounts
        # pad_height and pad_width are the amounts needed to reach config.grid_size
        pad_height = tf.maximum(0, self.config.grid_size - height)
        pad_width = tf.maximum(0, self.config.grid_size - width)

        # Pad only if needed (e.g., if current dim < target dim)
        padded_grid = tf.pad(
            grid_tensor,
            [[0, pad_height], [0, pad_width]],
            constant_values=0 # Use 0 for padding as it's a valid color and common background
        )

        # Crop if the original grid (or padded grid) is larger than the target size
        cropped_grid = padded_grid[:self.config.grid_size, :self.config.grid_size]

        return tf.cast(cropped_grid, dtype=tf.int32)

    def _get_data_generator(self):
        """Placeholder for a generator function, implemented by subclasses."""
        raise NotImplementedError("Subclasses must implement _get_data_generator.")

    def load_dataset(self):
        """Returns a batched, pre-processed tf.data.Dataset from the generator."""
        # Pass the method reference, not the result of calling the method
        ds = tf.data.Dataset.from_generator(
            self._get_data_generator,
            output_signature=tf.TensorSpec(shape=(None, None), dtype=tf.int32) # Grids of varying (H,W)
        )

        # Filter out empty or invalid grids early
        ds = ds.filter(lambda x: tf.rank(x) == 2 and tf.shape(x)[0] > 0 and tf.shape(x)[1] > 0)

        # Apply padding/cropping to a uniform size (e.g., 30x30)
        ds = ds.map(self._pad_grid_tf, num_parallel_calls=tf.data.AUTOTUNE)

        # --- Handle input representation (one-hot or single channel) ---
        if self.config.use_one_hot_input:
            # Cast to int64 for one_hot, then to float32 for model input
            ds = ds.map(lambda x: tf.one_hot(tf.cast(x, tf.int64), depth=self.config.num_colors, dtype=tf.float32),
                        num_parallel_calls=tf.data.AUTOTUNE)
            # Target shape for one-hot encoded input
            target_shape = tf.constant([self.config.grid_size, self.config.grid_size, self.config.num_colors], dtype=tf.int32)
        else:
            # Add a channel dimension (e.g., (H, W) -> (H, W, 1))
            ds = ds.map(lambda x: tf.expand_dims(x, axis=-1), num_parallel_calls=tf.data.AUTOTUNE)
            # Target shape for integer input
            target_shape = tf.constant([self.config.grid_size, self.config.grid_size, 1], dtype=tf.int32)
        # --- END Input Representation ---

        # Ensure and filter by target shape for robustness
        # Removed the problematic tf.ensure_shape call within map
        ds = ds.filter(lambda x: tf.reduce_all(tf.equal(tf.shape(x), target_shape)))

        ds = ds.shuffle(buffer_size=5000)
        ds = ds.batch(self.config.batch_size)
        ds = ds.prefetch(tf.data.AUTOTUNE)
        return ds

class ARCDataLoader(BaseDataLoader):
    """Data loader for the official ARC training challenges."""
    def _get_data_generator(self):
        """Loads grids from the official ARC training challenges JSON file."""
        try:
            with open(self.config.training_challenges_path, 'r') as f:
                full_data = json.load(f)
        except FileNotFoundError:
            logging.error(f"ARC training challenges file not found at: {self.config.training_challenges_path}")
            sys.exit(1) # Exit if essential file is missing
        except json.JSONDecodeError as e:
            logging.error(f"Error decoding JSON from {self.config.training_challenges_path}: {e}")
            sys.exit(1)

        logging.info(f"Loading ARC training challenges from {self.config.training_challenges_path}")
        for task_id, task_data in full_data.items():
            for example in task_data.get('train', []):
                if 'output' in example and example['output'] is not None:
                    # Convert list of lists to NumPy array
                    output_grid = np.array(example['output'], dtype=np.int32)
                    yield output_grid

class SyntheticDataLoader(BaseDataLoader):
    """Data loader for GAN-generated synthetic grids."""
    def _get_data_generator(self):
        """Loads grids from multiple synthetic dataset JSON files."""
        for path in self.config.synthetic_data_paths:
            try:
                with open(path, 'r') as f:
                    data = json.load(f)
                    logging.info(f"Loading synthetic data from {path}")
                    for item in data:
                        if isinstance(item, list) and all(isinstance(sublist, list) for sublist in item):
                            # Assume it's a list of lists (2D grid)
                            yield np.array(item, dtype=np.int32)
                        elif isinstance(item, list) and all(isinstance(val, (int, float)) for val in item):
                            # Assume it's a flattened list, try to reshape to a square
                            grid_size = int(np.sqrt(len(item)))
                            if grid_size * grid_size == len(item):
                                yield np.array(item, dtype=np.int32).reshape(grid_size, grid_size)
                            else:
                                logging.warning(f"Skipping malformed flattened grid in {path} due to non-square length: {len(item)}")
                        else:
                            logging.warning(f"Skipping malformed data item in {path}: {type(item)}")
            except (FileNotFoundError, json.JSONDecodeError) as e:
                logging.warning(f"Skipping file {path} due to error: {e}")
            except Exception as e:
                logging.error(f"An unexpected error occurred while loading {path}: {e}")

class Sampling(layers.Layer):
    """Uses (z_mean, z_log_var) to sample z, the vector encoding a digit."""
    def call(self, inputs):
        z_mean, z_log_var = inputs
        batch = tf.shape(z_mean)[0]
        dim = tf.shape(z_mean)[1]
        epsilon = tf.random.normal(shape=(batch, dim))
        return z_mean + tf.exp(0.5 * z_log_var) * epsilon

class Encoder(Model):
    def __init__(self, latent_dim, grid_size, name='encoder', **kwargs):
        super(Encoder, self).__init__(name=name, **kwargs)
        self.latent_dim = latent_dim
        self.grid_size = grid_size
        self.conv1 = layers.Conv2D(32, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn1 = layers.BatchNormalization()
        self.conv2 = layers.Conv2D(64, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn2 = layers.BatchNormalization()
        # Removed self.reshape_before_flatten as discussed
        self.flatten = layers.Flatten()
        self.z_mean_dense = layers.Dense(latent_dim, name='z_mean')
        self.z_log_var_dense = layers.Dense(latent_dim, name='z_log_var')

    def call(self, inputs, training=None):
        x = self.conv1(inputs)
        x = self.bn1(x, training=training)
        x = self.conv2(x)
        x = self.bn2(x, training=training)
        x = self.flatten(x) # Now directly flattening
        z_mean = self.z_mean_dense(x)
        z_log_var = self.z_log_var_dense(x)
        return z_mean, z_log_var

class Sampling(layers.Layer):
    def call(self, inputs):
        z_mean, z_log_var = inputs
        batch = tf.shape(z_mean)[0]
        dim = tf.shape(z_mean)[1]
        epsilon = tf.random.normal(shape=(batch, dim))
        return z_mean + tf.exp(0.5 * z_log_var) * epsilon

class Decoder(layers.Layer):
    def __init__(self, latent_dim, num_colors, grid_size, name='decoder', **kwargs):
        super(Decoder, self).__init__(name=name, **kwargs)
        self.num_colors = num_colors
        self.latent_dim = latent_dim

        # Calculate initial dimensions for deconvolution based on fixed grid_size
        # For grid_size=30, two stride-2 convs (padding='same') yield 8x8 spatial dims.
        # ceil(ceil(grid_size / 2) / 2)
        initial_h_float = tf.math.ceil(tf.cast(grid_size, tf.float32) / 2.0)
        initial_h_float = tf.math.ceil(initial_h_float / 2.0)
        self.initial_h = tf.cast(initial_h_float, tf.int32).numpy()
        self.initial_w = self.initial_h # Assuming square grids

        self.dense = layers.Dense(self.initial_h * self.initial_w * 16, activation='relu')
        self.reshape = layers.Reshape((self.initial_h, self.initial_w, 16))

        self.deconv1 = layers.Conv2DTranspose(64, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn1 = layers.BatchNormalization()
        self.deconv2 = layers.Conv2DTranspose(32, 3, activation='relu', strides=2, padding='same', kernel_initializer=GlorotNormal())
        self.bn2 = layers.BatchNormalization()

        self.output_conv = layers.Conv2DTranspose(num_colors, 3, activation='softmax', padding='same', kernel_initializer=GlorotNormal())

    def call(self, inputs):
        x = self.dense(inputs)
        x = self.reshape(x)
        x = self.deconv1(x)
        x = self.bn1(x)
        x = self.deconv2(x)
        x = self.bn2(x)
        return self.output_conv(x)

class VAE(Model):
    def __init__(self, latent_dim, num_colors, grid_size, name="vae", **kwargs):
        super(VAE, self).__init__(name=name, **kwargs)
        self.encoder = Encoder(latent_dim=latent_dim, grid_size=grid_size)
        self.sampling = Sampling()
        self.decoder = Decoder(latent_dim=latent_dim, num_colors=num_colors, grid_size=grid_size)
        self.total_loss_tracker = tf.keras.metrics.Mean(name="total_loss")
        self.reconstruction_loss_tracker = tf.keras.metrics.Mean(name="reconstruction_loss")
        self.kl_loss_tracker = tf.keras.metrics.Mean(name="kl_loss")

    def call(self, inputs, training=None):
        inputs_f32 = tf.cast(inputs, dtype=tf.float32)
        z_mean, z_log_var = self.encoder(inputs_f32, training=training) # Pass training to encoder
        z = self.sampling([z_mean, z_log_var])
        reconstruction = self.decoder(z)
        return reconstruction

    @property
    def metrics(self):
        return [
            self.total_loss_tracker,
            self.reconstruction_loss_tracker,
            self.kl_loss_tracker,
        ]

    def train_step(self, data):
        # Assuming `data` is just the input grid itself for a VAE
        # In a typical VAE, input == target for reconstruction loss

        with tf.GradientTape() as tape:
            inputs_f32 = tf.cast(data, dtype=tf.float32)
            z_mean, z_log_var = self.encoder(inputs_f32, training=True)
            z = self.sampling([z_mean, z_log_var])
            reconstruction = self.decoder(z)

            # Ensure data (target) and reconstruction are float32
            data_f32 = tf.cast(data, dtype=tf.float32)

            # Reconstruction loss (e.g., SparseCategoricalCrossentropy for classification per pixel)
            # Reshape reconstruction to (batch*H*W, num_colors)
            # Reshape data_f32 to (batch*H*W,)

            # Assuming `data` contains integer labels for colors (0-9)
            # The reconstruction output is (batch, H, W, num_colors) with softmax
            # Target `data` needs to be (batch, H, W) of integer labels

            # Flatten to calculate loss efficiently
            reconstruction_flattened = tf.reshape(reconstruction, [-1, self.decoder.num_colors])
            data_flattened = tf.reshape(tf.cast(data, tf.int32), [-1]) # Target labels as integers

            reconstruction_loss = tf.keras.losses.sparse_categorical_crossentropy(
                data_flattened, reconstruction_flattened, from_logits=False
            )
            reconstruction_loss = tf.reduce_mean(reconstruction_loss) # Mean over all pixels in batch

            # KL divergence loss
            kl_loss = -0.5 * (1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var))
            kl_loss = tf.reduce_mean(tf.reduce_sum(kl_loss, axis=1)) # Sum over latent_dim, then mean over batch

            total_loss = reconstruction_loss + kl_loss

        grads = tape.gradient(total_loss, self.trainable_weights)
        self.optimizer.apply_gradients(zip(grads, self.trainable_weights))

        self.total_loss_tracker.update_state(total_loss)
        self.reconstruction_loss_tracker.update_state(reconstruction_loss)
        self.kl_loss_tracker.update_state(kl_loss)
        return {
            "loss": self.total_loss_tracker.result(),
            "reconstruction_loss": self.reconstruction_loss_tracker.result(),
            "kl_loss": self.kl_loss_tracker.result(),
        }

    def test_step(self, data):
        inputs_f32 = tf.cast(data, dtype=tf.float32)
        z_mean, z_log_var = self.encoder(inputs_f32, training=False)
        z = self.sampling([z_mean, z_log_var])
        reconstruction = self.decoder(z)

        data_f32 = tf.cast(data, dtype=tf.float32)
        reconstruction_flattened = tf.reshape(reconstruction, [-1, self.decoder.num_colors])
        data_flattened = tf.reshape(tf.cast(data, tf.int32), [-1])

        reconstruction_loss = tf.keras.losses.sparse_categorical_crossentropy(
            data_flattened, reconstruction_flattened, from_logits=False
        )
        reconstruction_loss = tf.reduce_mean(reconstruction_loss)

        kl_loss = -0.5 * (1 + z_log_var - tf.square(z_mean) - tf.exp(z_log_var))
        kl_loss = tf.reduce_mean(tf.reduce_sum(kl_loss, axis=1))

        total_loss = reconstruction_loss + kl_loss

        self.total_loss_tracker.update_state(total_loss)
        self.reconstruction_loss_tracker.update_state(reconstruction_loss)
        self.kl_loss_tracker.update_state(kl_loss)
        return {
            "loss": self.total_loss_tracker.result(),
            "reconstruction_loss": self.reconstruction_loss_tracker.result(),
            "kl_loss": self.kl_loss_tracker.result(),
        }

def preprocess_arc_grid(grid_tensor, target_size):
    """
    Resizes, pads, or crops an input grid tensor to a fixed target_size,
    ensures it's float32, and adds a channel dimension if it's 2D.
    """
    # 1. Cast to float32
    grid_tensor = tf.cast(grid_tensor, dtype=tf.float32)

    # 2. Add channel dimension if it's currently (H, W)
    # tf.rank returns a tf.Tensor, which AutoGraph can handle in `if` statements
    if tf.rank(grid_tensor) == 2:
        grid_tensor = tf.expand_dims(grid_tensor, axis=-1) # Becomes (H, W, 1)

    # 3. Resize/Pad/Crop to the target_size
    # This is critical for fixed input shapes
    grid_tensor = tf.image.resize_with_crop_or_pad(
        grid_tensor,
        target_height=target_size,
        target_width=target_size
    )
    return grid_tensor

class PlausibilityValidator:
    """Uses the trained VAE to score the plausibility of an ARC grid."""
    def __init__(self, vae_model):
        self.vae_model = vae_model
        self.loss_fn = SparseCategoricalCrossentropy(from_logits=False)
    def get_plausibility_score(self, grid):
        """Calculates a plausibility score based on VAE's reconstruction error."""
        # Ensure input grid is int32 and add batch and channel dimensions for VAE input
        grid_tensor_input = tf.cast(tf.expand_dims(tf.expand_dims(grid, axis=0), axis=-1), dtype=tf.float32)
        reconstruction = self.vae_model(grid_tensor_input, training=False)

        # Prepare the target grid for loss calculation
        # Cast to int32, add batch dimension, and remove the extra channel dimension
        grid_tensor_loss_target = tf.cast(tf.expand_dims(grid, axis=0), dtype=tf.int32) # Shape (1, grid_size, grid_size)

        # Ensure shapes for robustness (optional, but good practice after manipulation)
        # reconstruction = tf.ensure_shape(reconstruction, (1, self.vae_model.grid_size, self.vae_model.grid_size, self.vae_model.num_colors))
        # grid_tensor_loss_target = tf.ensure_shape(grid_tensor_loss_target, (1, self.vae_model.grid_size, self.vae_model.grid_size))

        reconstruction_loss = self.loss_fn(grid_tensor_loss_target, reconstruction)
        return tf.exp(-reconstruction_loss).numpy()

class RLAgent:
    """
    A Reinforcement Learning agent using a Deep Q-Network (DQN) to generate
    plausible ARC grids based on feedback from a trained VAE.
    """
    def __init__(self, vae_model, grid_size, num_colors, learning_rate=0.001,
                 discount_factor=0.99, exploration_rate=1.0, min_exploration_rate=0.01,
                 exploration_decay_rate=0.995):
        self.vae_model = vae_model
        self.grid_size = grid_size
        self.num_colors = num_colors
        self.action_space_size = grid_size * grid_size * num_colors
        self.learning_rate = learning_rate
        self.discount_factor = discount_factor
        self.exploration_rate = exploration_rate
        self.min_exploration_rate = min_exploration_rate
        self.exploration_decay_rate = exploration_decay_rate
        self.q_network = self._build_q_network()
        self.optimizer = Adam(learning_rate=self.learning_rate)
    def _build_q_network(self):
        inputs = layers.Input(shape=(self.grid_size, self.grid_size, 1), dtype=tf.float32)
        x = layers.Conv2D(32, 3, activation='relu', padding='same')(inputs)
        x = layers.Flatten()(x)
        x = layers.Dense(128, activation='relu')(x)
        outputs = layers.Dense(self.action_space_size)(x)
        return Model(inputs=inputs, outputs=outputs)
    def choose_action(self, state_grid):
        if np.random.uniform(0, 1) < self.exploration_rate:
            action_index = np.random.randint(self.action_space_size)
        else:
            state_tensor = tf.cast(tf.expand_dims(tf.expand_dims(state_grid, 0), -1), dtype=tf.float32)
            q_values = self.q_network(state_tensor, training=False)
            action_index = tf.argmax(q_values[0]).numpy()
        return self._map_index_to_action(action_index)
    def _map_index_to_action(self, index):
        color = index % self.num_colors
        index = index // self.num_colors
        y = index % self.grid_size
        x = index // self.grid_size
        return (x, y, color)
    def _map_action_to_index(self, x, y, color):
        return x * self.grid_size * self.num_colors + y * self.num_colors + color
    def train_step(self, state, action, reward, next_state, done):
        state_tensor = tf.cast(tf.expand_dims(tf.expand_dims(state, 0), -1), dtype=tf.float32)
        next_state_tensor = tf.cast(tf.expand_dims(tf.expand_dims(next_state, 0), -1), dtype=tf.float32)
        with tf.GradientTape() as tape:
            q_values = self.q_network(state_tensor)
            if done:
                target_q_value = reward
            else:
                next_q_values = self.q_network(next_state_tensor)
                max_next_q_value = tf.reduce_max(next_q_values)
                target_q_value = reward + self.discount_factor * max_next_q_value
            action_mask = tf.one_hot(self._map_action_to_index(*action), self.action_space_size, dtype=tf.float32)
            q_value_for_action = tf.reduce_sum(q_values * action_mask, axis=1)
            loss = tf.keras.losses.MSE(tf.expand_dims(target_q_value, 0), q_value_for_action)
        grads = tape.gradient(loss, self.q_network.trainable_variables)
        self.optimizer.apply_gradients(zip(grads, self.q_network.trainable_variables))
    def get_vae_reward(self, grid):
        plausibility_validator = self.vae_model
        loss = -plausibility_validator.get_plausibility_score(grid)
        return float(loss)
    def decay_exploration_rate(self):
        self.exploration_rate = max(self.min_exploration_rate, self.exploration_rate * self.exploration_decay_rate)
    def generate_grid_with_rl(self, max_steps=1000):
        current_grid = np.zeros((self.grid_size, self.grid_size), dtype=np.int32)
        for step in range(max_steps):
            action = self.choose_action(current_grid)
            x, y, color = action
            next_grid = current_grid.copy()
            next_grid[x, y] = color
            reward = self.get_vae_reward(next_grid)
            done = step == max_steps - 1
            self.train_step(current_grid, action, reward, next_grid, done)
            current_grid = next_grid
        return current_grid
