setup_logging()

class VAESolver:
    """
    A class to handle the training and evaluation of the VAE model.
    """
    def __init__(self, model: ARC_VAE, config: dict, device: str = 'cpu', candidates_path: str | None = None, checkpoint_dir: str | None = None):
        self.model = model
        self.config = config
        self.device = torch.device(device)
        self.model.to(self.device)
        self.candidates_path = candidates_path
        self._candidates = None
        self.checkpointer = CheckpointHandler(checkpoint_dir) if checkpoint_dir else None
        
        self.optimizer = optim.Adam(
            self.model.parameters(),
            lr=self.config.get('learning_rate', 1e-3),
            weight_decay=self.config.get('weight_decay', 0.0)
        )
        self.kl_weight = self.config.get('kl_weight', 1.0)
        self.rl_guidance_weight = self.config.get('rl_guidance_weight', 0.0)

        if self.candidates_path:
            try:
                with open(self.candidates_path, 'r') as f:
                    self._candidates = json.load(f)
            except Exception:
                logger.warning(f"Could not load candidates from {self.candidates_path}")

    def vae_loss_function(self, recon_x, x, mu, logvar):
        """
        Calculates the VAE loss, which is the sum of the reconstruction loss
        and the KL divergence loss.
        
        Args:
            recon_x (torch.Tensor): The reconstructed grid from the decoder.
            x (torch.Tensor): The original grid.
            mu (torch.Tensor): The mean of the latent distribution.
            logvar (torch.Tensor): The log-variance of the latent distribution.
        
        Returns:
            torch.Tensor: The total VAE loss.
        """
        # Reconstruction loss (Cross-Entropy for one-hot encoded grids)
        recon_loss = nn.functional.cross_entropy(
            recon_x, 
            torch.argmax(x, dim=1), 
            reduction='mean'
        )
        
        # KL divergence loss
        kl_divergence = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp())
        
        total_loss = recon_loss + self.kl_weight * kl_divergence
        
        return total_loss, recon_loss, kl_divergence

    def train_step(self, data_loader: DataLoader, task_id: str | None = None, candidate_grid: np.ndarray | None = None):
        """
        Performs one epoch of training.
        """
        self.model.train()
        total_loss = 0
        for i, batch in enumerate(data_loader):
            input_tensors = batch['input'].to(self.device)
            self.optimizer.zero_grad()
            
            recon_batch, mu, logvar = self.model(input_tensors)
            
            loss, recon_loss, kl_loss = self.vae_loss_function(recon_batch, input_tensors, mu, logvar)

            # Hybrid guidance loss to pull reconstruction toward RL candidate
            if self.rl_guidance_weight > 0.0:
                cand = None
                if candidate_grid is not None:
                    cand = candidate_grid
                elif task_id is not None and isinstance(self._candidates, dict) and task_id in self._candidates:
                    cand = np.array(self._candidates[task_id], dtype=np.int64)
                if cand is not None:
                    # Build candidate target tensor matching input shape (N, C, H, W)
                    with torch.no_grad():
                        B, C, H, W = input_tensors.shape
                        target = torch.from_numpy(cand).long().to(self.device)
                        if target.shape != (H, W):
                            # Skip if mismatch; future: pad/crop
                            pass
                        else:
                            target = target.unsqueeze(0).repeat(B, 1, 1)
                            cand_loss = nn.functional.cross_entropy(recon_batch, target, reduction='mean')
                            loss = loss + self.rl_guidance_weight * cand_loss
            
            loss.backward()
            self.optimizer.step()
            total_loss += loss.item()
            
            if i % 100 == 0:
                logger.info(
                    f"  Batch {i}: Loss = {loss.item():.4f}, "
                    f"Recon Loss = {recon_loss.item():.4f}, "
                    f"KL Loss = {kl_loss.item():.4f}"
                )
                
        avg_loss = total_loss / len(data_loader)
        logger.info(f"End of training epoch. Average Loss = {avg_loss:.4f}")
        return avg_loss

    def evaluate(self, data_loader: DataLoader):
        """
        Evaluates the model on a validation or test dataset.
        """
        self.model.eval()
        total_loss = 0
        with torch.no_grad():
            for batch in data_loader:
                input_tensors = batch['input'].to(self.device)
                
                recon_batch, mu, logvar = self.model(input_tensors)
                
                loss, _, _ = self.vae_loss_function(recon_batch, input_tensors, mu, logvar)
                total_loss += loss.item()

        avg_loss = total_loss / len(data_loader)
        logger.info(f"Evaluation finished. Average Loss = {avg_loss:.4f}")
        return avg_loss

    def train_task(self, train_loader: DataLoader, val_loader: DataLoader | None, task_id: str, epochs: int):
        best_metric = float('inf')
        for epoch in range(epochs):
            logger.info(f"[Task {task_id}] Epoch {epoch+1}/{epochs}")
            train_loss = self.train_step(train_loader, task_id=task_id)
            metric = train_loss
            if val_loader is not None:
                val_loss = self.evaluate(val_loader)
                metric = val_loss
                logger.info(f"[Task {task_id}] Val Loss = {val_loss:.4f}")
            if self.checkpointer:
                # save latest
                self.checkpointer.save_named(self.model, f"vae_latest_task_{task_id}.pt", optimizer=self.optimizer)
                if metric < best_metric:
                    best_metric = metric
                    self.checkpointer.save_named(self.model, f"vae_best_task_{task_id}.pt", optimizer=self.optimizer)

    def solve(self, input_tensor: torch.Tensor):
        """
        Performs inference to reconstruct a single grid.
        
        Args:
            input_tensor (torch.Tensor): The preprocessed input grid.
            
        Returns:
            torch.Tensor: The reconstructed grid.
        """
        # Load best for task if available must be handled by caller via load_best_for_task
        self.model.eval()
        with torch.no_grad():
            input_tensor = input_tensor.unsqueeze(0).to(self.device) # Add batch dimension
            recon_grid, _, _ = self.model(input_tensor)
            return recon_grid.squeeze(0).cpu() # Remove batch dimension

    def load_best_for_task(self, task_id: str) -> bool:
        if not self.checkpointer:
            return False
        return self.checkpointer.load_named(self.model, f"vae_best_task_{task_id}.pt", self.device)

# Custom Dataset for VAE training
class VAEProblemDataset(Dataset):
    def __init__(self, data_pairs):
        self.data_pairs = data_pairs

    def __len__(self):
        return len(self.data_pairs)

    def __getitem__(self, idx):
        pair = self.data_pairs[idx]
        return {
            'input': pair['input'],
            'output': pair['output']
        }
