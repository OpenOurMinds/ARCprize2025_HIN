import numpy as np
import logging

from src.utils.logger_setup import setup_logging

setup_logging()
logger = logging.getLogger(__name__)


class ActionApplier:
    """
    Unified action applier operating on numpy arrays (H, W) of integer color ids.
    """
    def __init__(self, config: dict):
        self.config = config
        self.grid_size = tuple(config.get('grid_size', (3, 3)))
        self.num_colors = int(config.get('num_colors', 10))

    def apply(self, grid: np.ndarray, action_id: int) -> np.ndarray:
        if action_id == 0:
            return np.copy(grid)
        elif action_id == 1:
            return np.rot90(grid, k=1)
        elif action_id == 2:
            return np.rot90(grid, k=2)
        elif action_id == 3:
            return np.rot90(grid, k=3)
        elif action_id == 4:
            return np.fliplr(grid)
        elif action_id == 5:
            return np.flipud(grid)
        elif action_id == 6:
            # Fill with color 0 (toy)
            return np.zeros_like(grid)
        else:
            logger.warning(f"Unknown action_id {action_id}, returning copy")
            return np.copy(grid)
